	ex. 게시판 (10개 레코드)

		전체 페이지갯수: roundUp(10/3) = 4	int totalPage = Math.ceil(totalRecord/(float)3))
		현재 페이지번호:						 	int curPage = 1; <=== 초기값 1
		
		# 구현
		int curPage = 1;	// 현재페이지 번호
		int perPage = 3;	// 페이지당 보여줄 갯수
		int offset = (curPage - 1)*perPage	// 정렬된 레코드에서 레코드를 가져오기 위한 시작 위치
		List<EmpDTO> list  = session.selectList("",new RowBounds(int offset,perPage));

2. resultMap
	
	1) 용도 2가지 
	
		가. 테이블의 컬럼명과 DTO 의 변수명이 다른 경우
		
			<resultMap id="userResultMap" type="EmpDTO">
				<id property="DTO변수명" column="컬럼헤더값 or 별칭"/>
				<result property="username" column="ename"/>
			</resultMap>
			
				<select id="findByRseultMap" resultMap="userResultMap">
					select empno,
							ename,
							job,
							mgr,
							hiredate,
							sal,
							comm,
							deptno				
					from emp
					order by empno desc	
				</select>
				
		나. 조인할 때
			
			a. 1:1 관계 ( emp입장에서 dept 관계가 1:1 )
				
				# 자바코드 표현
				
				EmpDTO						DeptDTO
				
				int empno;					int deptno;
				string ename;				String dname;
				...							String loc;
				//int deptno;
				DeptDTO dto;
				
				# MyBatis
				
				<resultMap id="userResultMap" type="EmpDTO">
					<!-- property: type class의 멤버 변수명, column: 컬럼명 -->
					<!-- id => PK 설정 -->
					<id property="empno" column="empno"/>
					<result property="username" column="ename"/>
					<result property="job" column="job"/>
					<result property="mgr" column="mgr"/>
					<result property="hiredate" column="hiredate"/>
					<result property="salary" column="sal"/>
					<result property="commission" column="comm"/>					
					
					<!--Dept레코드 설정-->
					
					<association property="dto" javaType="DeptDTo">
					
					 <id property="deptno" column="xxx"/>
					 <result property="dname" column="yyy"/>
					 <result property="loc" column="zzz"/>
					
				  </association>
				</resultMap>
				
				<!-- emp와 dept 조인 -->
				<select id="" resultMap = "userResultMap">
					select empno,
							ename,
							job,
							mgr,
							hiredate,
							sal,
							comm,							
							dept.deptno	as xxx,
							dept.dname as yyy,
							dept.loc as zzz
							
					from emp join dept on emp.deptno = dept.deptno;
					order by empno desc	
				</select>
				
				
			b. 1:N 관계 ( dept입방에서는 emp 관계가 1:N)
						
				# 자바코드 표현
				
				EmpDTO						DeptDTO
				
				int empno;					int deptno;
				string ename;				String dname;
				...							String loc;
				//int deptno;				List<EmpDto> empList;	
												// dept 입장에서 deptno 하나에 EmpDTO의 여러 데이터라 딸려온다
												
				# MyBatis
				
					<resultMap id="empjoinMap2" type="DeptDTO">
						<id property="deptno" column="deptno"/>
						<result property="dname" column="dname"/>
						<result property="loc" column="loc"/>
					 
						<!-- List<EmpDTO> empList; 설정 -->
						<collection property="empList" oftype="EmpDTO">
							<id property="empno" column="empno"/>
							<result property="username" column="ename"/>
							<result property="job" column="job"/>
							<result property="mgr" column="mgr"/>
							<result property="hiredate" column="hiredate"/>
							<result property="salary" column="sal"/>
							<result property="commission" column="comm"/>		
						</collection>
					</resultMap>
					<select id="empdeptjoin2" resultMap="empjoinMap2">
						select  *
						from emp join dept on emp.deptno = dept.deptno
						order by empno desc
					</select>
