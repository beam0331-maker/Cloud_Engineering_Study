	2) schema.sql과 data.sql
		
		# src/main/resources/schema.sql
	
		# schema.sql
		
		creat table todo
		( id int auto_increment,
		  userid varchar(255) not null,
		  description varchar(255) not null,
		  target_date date not null,		  
		  # 자바는 카멜표기법 주로 사용하고, SQL은 스네이크표기법을 많이 사용한다
		  # targetDate <-------> target_date 자동으로 언더바 표기법의 컬럼을 카멜 표기법의 자바변수로 자동 매핑하도로 지원됨
		  done boolean,
		  primary key(id)
		);		  
		  
		CREATE TABLE member
		( userid VARCHAR(255),
		password VARCHAR(255) not null,
		username varchar(255) not null,
		role VARCHAR(255) DEFAULT 'USER' not null,
		PRIMARY KEY(userid)
		);

			// 로그인

	6) src/main/resources/com/exam/mapper/TodoMapper.xml
		src/main/java/com/exam/mapper/TodoMapper.java
		
		# TodoMapper.xml
		<?xml version="1.0" encoding="UTF-8" ?>
		<!DOCTYPE mapper
				  PUBLIC "-//mybatis.org//DTD Mapper 3.0//EN"
				  "https://mybatis.org/dtd/mybatis-3-mapper.dtd">
		<mapper namespace="com.exam.config.TodoMapper">

			 <select id="findAll" resultType="TodoDTO" parameterType="string">
				  select id, userid, description, target_date, done
				  from todo
				  where userid = #{userid}
			 </select>

			 <select id="findById" resultType="TodoDTO" parameterType="int">
				  select id, userid, description, target_date, done
				  from todo
				  where id = #{id}
			 </select>

			 <insert id="save" parameterType="TodoDTO">
				  insert into todo(userid, description, target_date, done)
				  values ( #{userid}, #{description}, #{target_date}, false)
			 </insert>

			 <update id="updateById" parameterType="TodoDTO">
				  update todo
				  set description = #{description},
						target_date = #{target_date},
						done = #{done}
				  where id = #{id}
			 </update>

			 <delete id="deleteById" parameterType="int">
				  delete from todo
				  where id = #{id}
			 </delete>

		</mapper>
		
		# TodoMapper.java
		@Mapper
		public interface TodoMapper {

			 List<TodoDTO> findAll(String userid);
			 TodoDTO findById(int id);
			 int save(TodoDTO todoDTO);
			 int updateById(TodoDTO todoDTO);
			 int deleteById(int id);
		}
