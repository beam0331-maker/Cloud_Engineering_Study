				a. select 태그
					
					// 단일행
					<select id = "findByDeptno" resultType="com/dto/DeptDTO" parameterType="int" >
						select deptno, dename, loc
						from dept 
						where deptno = #{deptno} 	=> ; 제외,
					</select>							#{변수} 대신에 진짜 데이터로 치환됨.
															JDBC의 ? 역할.
															${변수} 사용가능하지만 절대 사용불가
				
					# 자바코드에서 호출 하는 방법:
					SqlSession session = sqlSessionFactory.openSession();
					T t =  session.selectOne(String);		// String = 태그 id
					T t =  session.selectOne(String, Object);		// Object는 #{변수}에 입력할 값
					# T는 resultType 과 같아야 한다.
					
					ex.
						SqlSession session = sqlSessionFactory.openSession();
						DeptDTO dto = session.selectOne("findByDeptno", 10);
						
					// 다중행	
					<select id = "findByAll" resultType="com.dto.DeptDTO">
						select deptno, dname, loc
						from dept	
					</select>
					
					# 자바코드에서 호출 하는 방법:		
					
					SqlSession session = sqlSessionFactory.openSession();
					# E = resultType
					List<E> list = session.selectList(String); # String = 태그 id
					List<E> list = session.selectList(String, Object); # Object = #{변수} 값
					List<E> list = session.selectList(String,Object, RowBounds); 
					
					ex.
					SqlSession session = sqlSessionFactory.openSession();
					List<DeptDTO> list = session.selectList("findAll");
					
					
				속성
				id	 : 태그 식별용
				
				resultType :
				select문 실행된 후 결과 레코드 하나를 저장할 때 사용하는 클래스. 패키지명 필수.select된 결과가 자동으로 저장됨. (***********)
				컬럼헤더이름으로 된 set메서드를 DTO에서 호출해서 저장한다.
				 ==> 컬럼 헤더와 DTO의 변수명이 다르면 set메서드 호출이 안된다.

				전달된 값이 자동으로 #{변수}에 치환된다.
